9. 装备、饰品制造成功率
Mathf.RoundToInt方法的10f这个参数的位置，改动这个就行，默认是0.5f，数字越大越容易出金
public float GetCraftSuccessRate(int id)
{
    if (this.armor.transform.Find("CraftOut").GetComponent<ArmorIcon>().enabled && this.armor.transform.Find("CraftOut").GetComponent<ArmorIcon>().armorData != null && this.armor.transform.Find("CraftOut").GetComponent<ArmorIcon>().armorData.name != string.Empty)
    {
        return 30f + RePlayer.Instance.PlayerData.forgeSuccessRate * 100f - (float)(10 * this.armor.transform.Find("CraftOut").GetComponent<ArmorIcon>().armorData.lv) + (float)Mathf.RoundToInt(10f * (float)RePlayer.Instance.PlayerData.armorExp[id]) + RePlayer.Instance.PlayerData.craftRecipeSuccessRateAdd;
    }
    if (this.armor.transform.Find("CraftOut").GetComponent<DecorationIcon>().enabled && this.armor.transform.Find("CraftOut").GetComponent<DecorationIcon>().decorationData != null && this.armor.transform.Find("CraftOut").GetComponent<DecorationIcon>().decorationData.name != string.Empty)
    {
        return 30f + RePlayer.Instance.PlayerData.forgeSuccessRate * 100f - (float)(10 * (this.armor.transform.Find("CraftOut").GetComponent<DecorationIcon>().decorationData.rareLv - 1)) + (float)Mathf.RoundToInt(10f * (float)RePlayer.Instance.PlayerData.decorationExp[id - 2]) + RePlayer.Instance.PlayerData.craftRecipeSuccessRateAdd;
    }
    return 0f;
}
